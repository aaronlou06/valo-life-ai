/* src/components/TodayCard.jsx */

import React from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
export default function TodayCard({ card, onPrimary }) {
  return (
    <View style={styles.card}>
      <Text style={styles.title}>{card.title}</Text>
      {card.subtitle ? <Text style={styles.subtitle}>{card.subtitle}</Text> : null}
      {card.data && card.data.detail ? <Text style={styles.detail}>{card.data.detail}</Text> : null}
      <TouchableOpacity
        accessibilityRole="button"
        accessibilityLabel={card.primaryAction?.label || 'Card action'}
        onPress={() => onPrimary(card.primaryAction)}
        style={styles.button}
      >
        <Text style={styles.buttonText}>{card.primaryAction?.label || 'Open'}</Text>
      </TouchableOpacity>
    </View>
  );
}

/* src/hooks/useTodayCards.js */

import { useState, useEffect, useRef } from 'react';
import { useIsFocused } from '@react-navigation/native';
import ApiClient from '@/apiClient'; // adapt
import { trackEvent } from '@/services/telemetry';
export default function useTodayCards(date) {
  const [cards, setCards] = useState([]);
  const [loading, setLoading] = useState(false);
  const isFocused = useIsFocused();
  const startTsRef = useRef(null);
  async function load() {
    setLoading(true);
    startTsRef.current = Date.now();
    try {
      const res = await ApiClient.post('/api/today/cards', { date });
      const payload = res?.data ?? [];
      setCards(payload);
      trackEvent('today.cards_fetched', { count: payload.length, source: 'server', latencyMs: Date.now() - startTsRef.current });
    } catch (err) {
      console.warn('today cards fetch failed', err);
      // fallback: single motivation card
      setCards([{
        id: 'motivation-fallback',
        type: 'motivation',
        title: 'Your motivation',
        subtitle: 'Tap to read what you wrote at onboarding',
        data: { text: 'Remember why you started.' },
        primaryAction: { label: 'Open motivation', actionType: 'open_motivation' }
      }]);
      trackEvent('today.cards_fetched', { count: 1, source: 'fallback', latencyMs: Date.now() - startTsRef.current });
    } finally {
      setLoading(false);
    }
  }
  useEffect(() => {
    if (isFocused) load();
  }, [isFocused, date]);
  return { cards, loading, refresh: load };
}
/* src/screens/Today.tsx (integration sketch) */

import React, { useMemo, useState } from 'react';
import { View, Text, TouchableOpacity, ScrollView } from 'react-native';
import TodayCard from '@/components/TodayCard';
import useTodayCards from '@/hooks/useTodayCards';
import { trackEvent } from '@/services/telemetry';
import { useNavigation } from '@react-navigation/native';
export default function TodayScreen() {
  const navigation = useNavigation();
  const { cards, loading, refresh } = useTodayCards();
  const [screenStartTs] = useState(Date.now());
  function handlePrimaryCTA() {
    trackEvent('today.primary_cta_click', { from: 'today_screen' });
    navigation.navigate('CheckIn'); // or your check-in flow
  }
  function onCardPrimary(action) {
    const now = Date.now();
    trackEvent('today.time_to_first_action', { ms: now - screenStartTs });
    trackEvent('today.card_action_click', { cardType: action?.payload?.type || 'unknown', actionLabel: action?.label });
    // handle action types
    if (action.actionType === 'start_checkin') navigation.navigate('CheckIn', { focus: action.payload?.focus });
    if (action.actionType === 'open_goal') navigation.navigate('GoalDetail', { goalId: action.payload?.goalId});
    if (action.actionType === 'open_motivation') navigation.navigate('Profile', { screen: 'Motivation' });
  }
  return (
    <ScrollView contentContainerStyle={{ padding: 16 }}>
      <Text style={{ fontSize: 20, fontWeight: '700', marginBottom: 12 }}>Good morning</Text>
      <TouchableOpacity onPress={handlePrimaryCTA} style={{ backgroundColor: '#06B6D4', padding:14, borderRadius: 12, alignItems:'center', marginBottom: 16 }}>
        <Text style={{ color: 'white', fontWeight: '700' }}>Check in with Valo</Text>
      </TouchableOpacity>
      {loading ? <Text>Loading cards…</Text> : (
        cards.slice(0,3).map(c => <TodayCard key={c.id} card={c} onPrimary={onCardPrimary} />)
      )}
      <TouchableOpacity onPress={refresh} style={{ marginTop: 8 }}>
        <Text style={{ color: '#2563EB' }}>Refresh</Text>
      </TouchableOpacity>
    </ScrollView>
  );
}
