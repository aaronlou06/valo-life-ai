useTodayCards hook:

// hooks/useTodayCards.js
import { useState, useEffect } from 'react';
import { useIsFocused } from '@react-navigation/native';
import ApiClient from '@/api'; // adapt path
import { trackEvent } from '@/services/telemetry';
export default function useTodayCards(date) {
  const [cards, setCards] = useState([]);
  const [loading, setLoading] = useState(false);
  const isFocused = useIsFocused();
  async function load() {
    setLoading(true);
    try {
      const res = await ApiClient.post('/api/today/cards', { date });
      setCards(res.data || []);
      trackEvent('today.cards_fetched', { count: (res.data||[]).length });
    } catch (e) {
      console.warn('today cards fetch error', e);
    } finally {
      setLoading(false);
    }
  }
  useEffect(() => {
    if (isFocused) load();
  }, [isFocused, date]);
  return { cards, loading, refresh: load };
}
